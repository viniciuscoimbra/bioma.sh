Quickstart

Primeira análise em 30 segundos. Endpoint base https://api.refy.app/v1.

POST /v1/analyses 
# crie uma análise
curl https://api.refy.app/v1/analyses \
  -H "Authorization: Bearer refy_live_sk_..." \
  -H "Content-Type: application/json" \
  -d '{
    "url": "https://globoeditorial.com/post/o-que-e-ia",
    "market": "BR",
    "model": "refy-pro",
    "webhook": "https://hooks.globoeditorial.com/refy"
  }'

# resposta
{
  "id": "an_01HKZQ7...",
  "status": "queued",
  "credits_estimated": 3,
  "webhook_signature": "whsec_..."
}
